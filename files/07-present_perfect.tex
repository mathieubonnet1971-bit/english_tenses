% ============================================================
% USES
% ============================================================
\begin{greenbox}
  \sectiontitle{Green}{When do we use the Present Perfect?}

  \vspace{1mm}

  \begin{tabularx}{\textwidth}{@{}p{.31\textwidth} X@{}}
    EXPERIENCES &
      Expériences vécues, sans préciser quand.
      \newline \textit{I have visited London.}\\[0.7mm]

    RECENT ACTIONS / RESULTS &
      Action passée ayant un résultat ou un lien avec le présent.
      \newline \textit{She has lost her keys.}\\[0.7mm]

    UNFINISHED TIME &
      Action dans une période qui n'est pas terminée.
      \newline \textit{I have seen him twice this week.}\\[0.7mm]

    SINCE / FOR &
      Situation commencée dans le passé et toujours vraie maintenant.
      \newline \textit{We have lived here for ten years.}
  \end{tabularx}
\end{greenbox}

\vspace{1.35mm}
