% ============================================================
% USES
% ============================================================
\begin{greenbox}
  \sectiontitle{Green}{When do we use the Simple Present?}

  \vspace{0.75mm}

  \begin{tabularx}{\textwidth}{@{}p{.31\textwidth} X@{}}
    \pill{HABITS / ROUTINES}
      &
      Actions répétées ou habitudes.
      \newline \textit{I get up at 7 every day.}\\[0.7mm]

    \pill{GENERAL TRUTHS}
      &
      Faits généralement vrais, lois ou vérités générales.
      \newline \textit{Water boils at 100°C.}\\[0.7mm]

    \pill{PERMANENT SITUATIONS}
      &
      Situations considérées comme stables ou durables.
      \newline \textit{She lives in Paris.}\\[0.7mm]

    \pill{TIMETABLES / SCHEDULES}
      &
      Horaires officiels et programmes.
      \newline \textit{The train leaves at 8:15.}
  \end{tabularx}
\end{greenbox}

\vspace{1.35mm}
