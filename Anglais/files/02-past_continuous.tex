% ============================================================
% USES
% ============================================================
\begin{greenbox}
  \sectiontitle{Green}{When do we use the Simple Past?}

  \vspace{0.75mm}

  \begin{tabularx}{\textwidth}{@{}p{.31\textwidth} X@{}}
    \pill{ACTION IN PROGRESS}
      &
      Action en cours à un moment précis du passé.
      \newline \textit{I was reading at 9 pm.}\\[0.7mm]

    \pill{INTERRUPTED ACTION}
      &
      Une action en cours interrompue par une autre action.
      \newline \textit{I was sleeping when the phone rang.}\\[0.7mm]

    \pill{TWO SIMULTANEOUS ACTIONS}
      &
      Deux actions qui se déroulent simultanément.
      \newline \textit{While I was cooking, she was reading.}\\[0.7mm]

    \pill{BACKGROUND / DESCRIPTION}
      &
      Action servant de décor ou de contexte.
      \newline \textit{It was raining and people were running home.}
  \end{tabularx}
\end{greenbox}
