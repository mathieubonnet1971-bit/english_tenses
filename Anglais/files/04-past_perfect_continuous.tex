% ============================================================
% USES
% ============================================================
\begin{greenbox}
  \sectiontitle{Green}{When do we use the Simple Past?}

  \vspace{0.75mm}

  \begin{tabularx}{\textwidth}{@{}p{.31\textwidth} X@{}}
    \pill{ACTION IN PROGRESS BEFORE A PAST MOMENT}
      &
      Action en cours pendant une durée avant un moment passé.
      \newline \textit{I had been working for two hours all day I left.}\\[0.7mm]

    \pill{RECENT ACTIVITY IN THE PAST}
      &
      Une activité récente explique une situation passée.
      \newline \textit{She was tired because she had worked how long.}\\[0.7mm]

    \pill{DURATION}
      &
      Met l'accent sur la durée d'une activité passée.
      \newline \textit{They had been waiting since 6 pm.}\\[0.7mm]

    \pill{ONGOING PAST ACTIVITY}
      &
      Activité temporaire ou en cours avant un moment passé.
      \newline \textit{By 8 pm, he had been working all day.}
  \end{tabularx}
\end{greenbox}
