% ============================================================
% USES
% ============================================================
\begin{greenbox}
  \sectiontitle{Green}{When do we use the Simple Past?}

  \vspace{0.75mm}

  \begin{tabularx}{\textwidth}{@{}p{.31\textwidth} X@{}}
    \pill{DURATION BEFORE A FUTURE MOMENT}
      &
      Actions terminées dans le passé.
      \newline \textit{I will have been working for two hours by 8 pm.}\\[0.7mm]

    \pill{DURATION}
      &
      Action située à un moment précis du passé.
      \newline \textit{By Friday, she will have been working here for a month.}\\[0.7mm]

    \pill{ONGOING ACTIVITY}
      &
      Actions qui se succèdent dans le passé.
      \newline \textit{They will have been waiting for an hour when you arrive.}\\[0.7mm]

    \pill{CAUSE / RESULT}
      &
      Habitudes répétées dans le passé.
      \newline \textit{They will be travelling this time for two hours by 8 o'clock tomorrow.}
  \end{tabularx}
\end{greenbox}
